% pseudo-3D: two offset faces behind a node, one light direction for all blocks
\newcommand{\pseudo}[1]{%
  \begin{scope}[on background layer]
    \fill[black!20] ($(#1.south east)+(0.16,0.11)$) -- ($(#1.north east)+(0.16,0.11)$)
      -- (#1.north east) -- (#1.south east) -- cycle;
    \fill[black!10] ($(#1.north west)+(0.16,0.11)$) -- ($(#1.north east)+(0.16,0.11)$)
      -- (#1.north east) -- (#1.north west) -- cycle;
    \draw[line width=0.35pt, black!55] ($(#1.north west)+(0.16,0.11)$)
      -- ($(#1.north east)+(0.16,0.11)$) -- ($(#1.south east)+(0.16,0.11)$);
    \draw[line width=0.35pt, black!55] (#1.north east) -- ($(#1.north east)+(0.16,0.11)$);
    \draw[line width=0.35pt, black!55] (#1.south east) -- ($(#1.south east)+(0.16,0.11)$);
  \end{scope}%
}
